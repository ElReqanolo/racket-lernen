define-record – alle Formen

; ohne Prädikat
(define-record wallclock-time
  make-wallclock-time                  ; Konstruktor
  (wallclock-time-hour hour)           ; (Selektor Signatur) pro Feld
  (wallclock-time-minute minute))

; mit Prädikat (nötig für gemischte Daten)
(define-record dillo
  make-dillo
  dillo?                               ; Prädikat
  (dillo-weight natural)
  (dillo-alive? boolean))

; ohne Felder (genau ein Wert, z. B. für "kein Ergebnis")
(define-record no-result make-no-result no-result?)

; mit Signatur-Parametern (wie list-of)
(define-record (node-of leaf label)
  make-node
  node?
  (node-label label)
  (node-left-branch (tree-of leaf label))
  (node-right-branch (tree-of leaf label)))

; Liste als Eingabe
(define f
  (lambda (list)
    (cond
      ((empty? list) ...)
      ((cons? list) ... (first list) ... (f (rest list)) ...))))

; natürliche Zahl als Eingabe
(define f
  (lambda (n)
    (cond
      ((zero? n) ...)
      ((positive? n) ... (f (- n 1)) ...))))

; Liste mit Akkumulator
(define f
  (lambda (list0)
    (define accumulate
      ; Invariante: acc ist ... der Elemente zwischen list0 und list
      (lambda (list acc)
        (cond
          ((empty? list) acc)
          ((cons? list) (accumulate (rest list) (... (first list) acc))))))
    (accumulate list0 ...)))
